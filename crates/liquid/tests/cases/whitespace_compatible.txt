## keeps its first character when the tag after it trims it.
#! parse {"bug_compatible_whitespace_trimming": true}
a {{- 'b' -}} c
a   {{- 'b' }}   c
a \n {{- 'b' -}} \n c
{% if true %}\n  {{- 'b' }}\n{% endif %}
{% if true %}  \n{{- 'b' }}{% endif %}
{% if true %}\t\n {{- 'b' }}{% endif %}
{% if true %}\n\n\t{%- assign x = 1 %}{{ x }}{% endif %}
{{ 'a' }}\n    {{- 'b' }}\n    {{- 'c' }}
{{ 'a' -}}\n    {{- 'b' }}
{{ 'a' }}\n{%- if true -%}\n b \n{%- endif -%}\nc
\n{{- 'a' }}

## A robots.txt template, as themes write them.
#! data {"groups": [{"agent": "User-agent: *", "rules": ["Disallow: /a", "Disallow: /b"], "sitemap": "Sitemap: /sitemap.xml"}, {"agent": "User-agent: bot", "rules": ["Disallow: /"]}]}
# robots\n{% for group in groups %}\n  {{- group.agent }}\n  {%- for rule in group.rules -%}\n    {{- rule }}\n  {%- endfor -%}\n\n  {%- if group.agent == 'User-agent: *' -%}\n    {{ ' ' }}\n    {{- 'Disallow: /x' }}\n    {{- 'Disallow: /y' }}\n  {%- endif -%}\n\n  {%- if group.sitemap != blank -%}\n    {{- group.sitemap }}\n  {% endif %}\n{%- endfor -%}\n
# robots\n{%- comment -%}\n# Caution\n{% endcomment %}\n{% for group in groups %}\n  {{- group.agent -}}\n\n  {% for rule in group.rules %}\n    {{- rule -}}\n  {% endfor %}\n\n  {%- if group.sitemap != blank -%}\n    {{ group.sitemap }}\n  {%- endif -%}\n{% endfor %}\n
## The same block ends and output tags as above, under the storefront rule: whitespace-only text before a
## `{%-` keeps its first character (the capture of a form field's id ends with a line break, as the
## gem prints it with the option and as a Shopify storefront printed it on 2026-10-20). Text that is
## not only whitespace is trimmed in full.
{% capture id -%}\n    field_{{ a }}_{{ b }}\n  {%- endcapture %}[{{ id }}]
{% capture id %}x {{ 'y' }} {%- endcapture %}[{{ id }}]
{% capture id %}x {{ 'y' }}\n {%- endcapture -%}\n[{{ id }}]
{% capture id %}x text {%- endcapture %}[{{ id }}]
{% capture id %}{% assign q = 1 %} {{ q }}  {%- endcapture %}[{{ id }}]
